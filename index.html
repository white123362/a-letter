import random import time from dataclasses import dataclass, field from typing import List, Dict, Optional # ---------- 数据模型 ---------- @dataclass class Character: name: str age: int gender: str # "男" / "女" personality: str occupation: str relationship_status: str = "已分手" # 已婚 / 已分手 / 分居中 purpose: str = "复合" # 复合 / 告别 / 寻找答案 issues: str = "" partner: Optional['Character'] = None affection: Dict[str, int] = field(default_factory=dict) # 对其他人好感度 is_alive: bool = True @dataclass class Couple: char_a: Character char_b: Character def __post_init__(self): self.char_a.partner = self.char_b self.char_b.partner = self.char_a self.char_a.affection[self.char_b.name] = 50 self.char_b.affection[self.char_a.name] = 50 # ---------- 全局状态 ---------- class GameState: def __init__(self, characters: List[Character], couples: List[Couple]): self.characters = characters self.couples = couples self.day = 0 self.will_cards: Dict[str, List[str]] = {c.name: [] for c in characters} self.mood_board: Dict[str, List[str]] = {c.name: [] for c in characters} self.ended = False self.final_choices: Dict[str, bool] = {} def advance_day(self): self.day += 1 def record_will(self, name: str, choice: str): if name in self.will_cards: self.will_cards[name].append(choice) def record_mood(self, name: str, mood: str): if name in self.mood_board: self.mood_board[name].append(mood) def get_mood_summary(self) -> Dict[str, int]: summary = {} for name, moods in self.mood_board.items(): if moods: summary[name] = moods[-1] # 只显示最新心情 return summary # ---------- 数据生成 ---------- def create_default_characters(): # 女方角色（沿用你的设定） females = [ Character("望舒", 25, "女", "表面温柔，内心阴暗", "游戏博主", "已分手", "复合"), Character("梦汝", 26, "女", "外表柔弱，控制欲极强", "舞者", "已分手", "告别"), Character("思瑶", 25, "女", "清纯人设，内心阴狠", "歌手", "已分手", "复合"), Character("郁瑾", 24, "女", "仁心表象，私心极重", "外科医生", "已分手", "复合"), Character("桃子", 24, "女", "温文尔雅，心胸狭隘", "声乐老师", "已分手", "复合"), ] # 男方角色（随机生成 5 位） male_names = ["紫苑", "姜赫", "陆屿", "周野", "谢衍"] personalities = ["沉默", "急躁", "温柔但懦弱", "精明", "自负"] occupations = ["CEO", "摄影师", "律师", "自由职业", "设计师"] males = [] for i, name in enumerate(male_names): males.append(Character( name=name, age=random.randint(24, 30), gender="男", personality=personalities[i % len(personalities)], occupation=occupations[i % len(occupations)], relationship_status="已分手", purpose=random.choice(["复合", "告别", "寻找答案"]), issues="性格不合 / 沟通问题 / 前任介入" if i > 0 else "发现自己不适合谈恋爱" )) return females, males def pair_couples(females: List[Character], males: List[Character]) -> List[Couple]: random.shuffle(males) # 随机配对 couples = [] for f, m in zip(females, males): # 给双方设置性格标签，方便剧情生成 f.issues = f"{m.name} 太闷了，不够懂我" if random.random() > 0.5 else "对未来规划不一致" m.issues = f"{f.name} 控制欲太强" if random.random() > 0.5 else "经常吵架，累了" couples.append(Couple(f, m)) return couples # ---------- 每日流程 ---------- def day_loop(state: GameState): print(f"\n{'='*50}") print(f"第 {state.day} 天") print(f"{'='*50}") # 显示当前CP print("\n【当前配对】") for c in state.couples: print(f" {c.char_a.name} ♥ {c.char_b.name}") # 意愿卡填写（模拟每个角色） print("\n【意愿卡填写】") for char in state.characters: choice = random.choice(["是", "否", "不确定"]) state.record_will(char.name, choice) print(f" {char.name}：{choice}") # 心情晴雨表 print("\n【心情晴雨表】") moods = ["😊", "😐", "😔", "😡", "😢"] for char in state.characters: mood = random.choice(moods) state.record_mood(char.name, mood) print(f" {char.name}：{mood}") # 随机事件（模拟剧情推进） trigger_random_event(state) # 采访环节 print("\n【采访】") for char in random.sample(state.characters, min(3, len(state.characters))): print(f" Q：{char.name}，今天有什么想对对方说的？") print(f" A：……（{char.name} 沉默了片刻，没有回答）") # 检查死亡/退出条件 check_game_over(state) def trigger_random_event(state: GameState): events = [ "你们在营地散步时，遇到了前任。", "有人酒后吐真言，爆出了旧事。", "下雨了，你和她共撑一把伞。", "节目组安排了任务：互换身份演戏。", "其中一位嘉宾情绪崩溃，当场离场。", "有人在房车里发现了一封旧情书。", "你收到了一条匿名消息。", ] print(f"\n【随机事件】{random.choice(events)}") def check_game_over(state: GameState): # 模拟死亡触发（随机概率） if random.random() < 0.05 and state.day > 10: victim = random.choice(state.characters) victim.is_alive = False print(f"\n⚠️ 突发悲剧：{victim.name} 因意外/冲突死亡。") state.ended = True # ---------- 抉择日 ---------- def final_choice_day(state: GameState): print("\n" + "="*50) print("第17天 · 最终抉择") print("="*50) for couple in state.couples: a = couple.char_a b = couple.char_b # 模拟各自决定（根据好感度+随机） a_decide = random.choice([True, False]) b_decide = random.choice([True, False]) state.final_choices[a.name] = a_decide state.final_choices[b.name] = b_decide print(f"\n【{a.name} & {b.name}】") if a_decide and b_decide: print(" ✅ 双方下车 → 复合/重新开始") elif a_decide or b_decide: print(" ⚠️ 一方下车 → 一方想挽回，一方已离开") else: print(" ❌ 双方不下车 → 各自分开") def show_final_summary(state: GameState): print("\n" + "="*50) print("【最终统计】") print("="*50) print("\n意愿卡趋势（最后3天）:") for name, cards in state.will_cards.items(): if len(cards) >= 3: print(f" {name}: {cards[-3:]}") print("\n心情晴雨表:") for name, moods in state.mood_board.items(): if moods: print(f" {name}: {moods[-1]}") # ---------- 主循环 ---------- def main(): print("\n" + "="*50) print("【恋综模拟器 · 18天旅程】") print("="*50) females, males = create_default_characters() couples = pair_couples(females, males) state = GameState(females + males, couples) # 第0天 · 前期拍摄 print("\n第0天 · 前期拍摄") for char in state.characters: print(f" {char.name}（{char.age}岁 · {char.occupation}）：{char.personality}") # 第1-16天 for _ in range(1, 17): if state.ended: break state.advance_day() day_loop(state) time.sleep(0.3) # 为了让输出可读 # 第17天 · 抉择 final_choice_day(state) # 第18天 · 返程 + 总结 print("\n第18天 · 返程/后采") show_final_summary(state) print("\n【旅程结束】") if __name__ == "__main__": main()
